169 | public export 0
170 | Pjrt : Type -> Type
171 | Pjrt = EitherT PjrtError IO
172 |
173 | try : AnyPtr -> AnyPtr -> a -> Pjrt a
174 | try api err onOk = if (isNullPtr err) then right onOk else do
175 |   msg <- pjrtErrorMessage api err
176 |   args <- primIO $ prim__mkPjrtErrorGetCodeArgs err
177 |   getCodeErr <- primIO $ prim__pjrtErrorGetCode api args
178 |   code <- if (isNullPtr getCodeErr) then pure Nothing else do
179 |     let code = prim__pjrtErrorGetCodeArgsCode args
180 |     destroyPjrtError api getCodeErr
181 |     pure $ Just code
182 |   primIO $ prim__deletePjrtErrorGetCodeArgs args
183 |   destroyPjrtError api err
184 |   left $ MkPjrtError msg $ map pjrtErrorCodeFromCInt code
