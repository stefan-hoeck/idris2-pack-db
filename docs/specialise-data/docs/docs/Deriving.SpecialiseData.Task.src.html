 60 | ||| Given a list of arguments and a sorted set of names,
 61 | ||| assert that every argument's name is in that set
 62 | checkArgsUse : MonadError SpecialisationError m => List Arg -> SortedSet Name -> m ()
 63 | checkArgsUse [] _ = pure ()
 64 | checkArgsUse (x :: xs) t = do
 65 |   let Just n = x.name
 66 |   | _ => checkArgsUse xs t
 67 |   if contains n t
 68 |     then checkArgsUse xs t
 69 |     else throwError UnusedVarError
 70 |
 71 | ||| Given a list of arguments and a list of their aliases, apply aliases to then
 72 | applyArgAliases :
 73 |   (as : List Arg) ->
 74 |   (0 _ : All IsNamedArg as) =>
 75 |   List (Name, Name) ->
 76 |   SortedMap Name TTImp ->
 77 |   Subset (List Arg) (All IsNamedArg)
 78 | applyArgAliases []        @{[]}     _  _   = Element [] []
 79 | applyArgAliases (x :: xs) @{_ :: _} ys ins = do
 80 |   let (newIns, newName, ys) : (SortedMap _ _, Name, List (Name, Name)) =
 81 |     case ys of
 82 |        []              => (ins                  , argName x, [])
 83 |        ((y, y') :: ys) => (insert y (var y') ins, y'       , ys)
 84 |   let Element rec prec = applyArgAliases xs ys newIns
 85 |   Element
 86 |     (MkArg x.count x.piInfo (Just newName) (substituteVariables newIns x.type) :: rec)
 87 |     (ItIsNamed :: prec)
 88 |
 89 | ||| Given a list of arguments, generate a list of aliased arguments
 90 | ||| and a list of aliases
 91 | transformArgNames :
 92 |   (f : Name -> Name) ->
 93 |   (as : List Arg) ->
 94 |   (0 _ : All IsNamedArg as) =>
 95 |   (Subset (List Arg) (All IsNamedArg), List (Name, Name))
 96 | transformArgNames f as = do
 97 |   let aliases = pushIn as %search <&> \(x `Element` xN) => (argName x, f $ Expr.argName x @{xN})
 98 |   (applyArgAliases as aliases empty, aliases)

100 | inGenNSImpl : Namespace -> Name -> Name -> Name
101 | inGenNSImpl (MkNS strs) p n = do
102 |   let newNS =
103 |     case n of
104 |         (NS (MkNS subs) n) => subs
105 |         n => []
106 |   NS (MkNS $ newNS ++ show p :: strs) $ dropNS n

111 | inGenNS task = inGenNSImpl task.currentNs task.resultName

114 | ||| Get all the information needed for specialisation from task
115 | export
116 | getTask :
117 |   Monad m =>
118 |   NamespaceProvider m =>
119 |   MonadError SpecialisationError m =>
120 |   NamesInfoInTypes =>
121 |   (resultName : Name) ->
122 |   (resultKind : TTImp) ->
123 |   (resultContent : TTImp) ->
124 |   m SpecTask
125 | getTask resultName resultKind resultContent = do
126 |   let (tqArgs, tqRet) = unLambda resultContent
127 |   -- Check for unused arguments
128 |   checkArgsUse tqArgs $ usesVariables tqRet
129 |   -- Extract name of polymorphic type
130 |   let (IVar _ typeName, _) = Expr.unAppAny tqRet
131 |   | _ => throwError TaskTypeExtractionError
132 |   -- Prove that all spec lambda arguments are named
133 |   let Yes tqArgsNamed = all isNamedArg tqArgs
134 |   | _ => throwError UnnamedArgInLambdaError
135 |   -- Create aliases for spec lambda's arguments and perform substitution
136 |   let (Element tqArgs tqArgsNamed, tqAlias) = transformArgNames (prependS "fv^\{resultName}^") tqArgs
137 |   let tqRet = substituteVariables (fromList $ mapSnd var <$> tqAlias) tqRet
138 |   let (ttArgs, _) = unPi resultKind
139 |   -- Check for partial application in spec
140 |   let True = (length tqArgs == length ttArgs)
141 |   | _ => throwError PartialSpecError
142 |   -- Prove that all spec lambda type's arguments are named
143 |   let Yes ttArgsNamed = all isNamedArg ttArgs
144 |   | _ => throwError UnnamedArgInLambdaError
145 |   -- Apply aliasing to spec lambda type's info
146 |   let Element ttArgs ttArgsNamed = applyArgAliases ttArgs tqAlias empty
147 |   -- Get current namespace
148 |   currentNs <- provideNS
149 |   -- Get polymorphic type's info
150 |   let Just polyTy = lookupType typeName
151 |   | _ => throwError $ MissingTypeInfoError typeName
152 |   -- Prove all its arguments/constructors/constructor arguments are named
153 |   let Yes polyTyNamed = areAllTyArgsNamed polyTy
154 |     | No _ => throwError $ UnnamedArgInPolyTyError polyTy.name
155 |   let specInvocation = reAppAny
156 |           (var (inGenNSImpl currentNs (snd $ unNS $ resultName) (snd $ unNS $ resultName))) $
157 |             ttArgs.appsWith @{ttArgsNamed} var empty
158 |   pure $ MkSpecTask
159 |     { tqArgs
160 |     , tqRet
161 |     , tqArgsNamed
162 |     , ttArgs
163 |     , ttArgsNamed
164 |     , currentNs
165 |     , resultName = snd $ unNS resultName
166 |     , fullInvocation = tqRet --- TODO: intelligent full invocation
167 |     , specInvocation
168 |     , polyTy
169 |     , polyTyNamed
170 |     }
